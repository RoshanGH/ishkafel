/// 时间线视口（不可变）：缩放与滚动状态 + ms↔px 换算
class TimelineGeometry {
  final int durationMs;
  final double msPerPx;     // 缩放：每像素毫秒数（越小越放大）
  final double scrollPx;    // 水平滚动（像素）
  const TimelineGeometry({required this.durationMs, required this.msPerPx, this.scrollPx = 0});

  double msToPx(int ms) => ms / msPerPx - scrollPx;
  int pxToMs(double px) => ((px + scrollPx) * msPerPx).round().clamp(0, durationMs);
  double get totalWidthPx => durationMs / msPerPx;

  TimelineGeometry copyWith({double? msPerPx, double? scrollPx});

  /// 以锚点像素位置为中心缩放（zoomFactor>1 放大），scroll 同步补偿，并 clamp 合法范围
  TimelineGeometry zoomAt(double anchorPx, double zoomFactor, {required double viewportWidthPx});

  /// 滚动并 clamp 到 [0, max(0, totalWidth - viewport)]
  TimelineGeometry scrolledBy(double deltaPx, {required double viewportWidthPx});

  /// 适配整个时长到给定视口宽（初始状态）
  static TimelineGeometry fit({required int durationMs, required double viewportWidthPx});

  /// 刻度间隔选择：返回不小于 minLabelSpacingPx 像素间距的"整"毫秒步长（1s/2s/5s/10s/30s/60s 档位）
  int rulerStepMs({double minLabelSpacingPx = 80});
}
```

/// 台词拆分：句子中点 < splitMs 的句子归左段
class TranscriptSplitter {
  static (String left, String right) splitAt(List<AsrSentence> sentences, int splitMs);
}

/// 切分编辑纯函数集。所有操作：帧对齐→约束校验→返回新列表；非法返回 null。
/// 不变量（每个操作输出必须满足）：单元无缝覆盖 [0,durationMs]；镜头无缝覆盖所属单元；
/// 首/末镜头边界==单元边界；一切边界为帧点；任何段时长 ≥ 一帧。
abstract final class SegmentationEditOps {
  static int frameMs(double fps) => (1000 / fps).round();

  /// 移动单元 i 与 i+1 之间的边界；两侧首尾镜头联动裁剪（被越过的镜头边界吞并）
  static List<SemanticUnit>? moveUnitBoundary(
      List<SemanticUnit> units, int i, int rawMs,
      {required double fps});

  /// 移动单元 u 内镜头 s 与 s+1 之间的边界（限制在两镜头内部）
  static List<SemanticUnit>? moveShotBoundary(
      List<SemanticUnit> units, int u, int s, int rawMs,
      {required double fps});

  /// 在 rawMs 处把单元 u 拆成两个（镜头随拆分点切开；台词按 sentences 分配）
  static List<SemanticUnit>? splitUnitAt(
      List<SemanticUnit> units, int u, int rawMs,
      {required double fps, required List<AsrSentence> sentences});

  /// 单元 u 并入前一单元（镜头列表拼接，原单元边界保留为镜头边界；台词拼接；tags 取并集）
  static List<SemanticUnit>? mergeUnitWithPrevious(List<SemanticUnit> units, int u);

  /// 在 rawMs 处把单元 u 内包含该点的镜头拆成两个
  static List<SemanticUnit>? splitShotAt(
      List<SemanticUnit> units, int u, int rawMs, {required double fps});

  /// 单元 u 内镜头 s 并入前一镜头（tags 取并集）
  static List<SemanticUnit>? mergeShotWithPrevious(List<SemanticUnit> units, int u, int s);

  /// 更新单元台词
  static List<SemanticUnit> updateTranscript(List<SemanticUnit> units, int u, String text);

  /// 校验不变量（供测试与调试断言用）
  static bool holdsInvariants(List<SemanticUnit> units, int durationMs, double fps);
}
```

联动裁剪规则（moveUnitBoundary 到新边界 b）：
- 左单元 i：保留 `startMs < b` 的镜头；最后保留镜头 `endMs = b`；若全被吞（b ≤ 首镜头 start——被 clamp 阻止，理论不发生）保底单镜头 `[startMs, b]`
- 右单元 i+1：保留 `endMs > b` 的镜头；首保留镜头 `startMs = b`；同理保底
- clamp 范围：`[units[i].startMs + frame, units[i+1].endMs - frame]`

   绘制层现已对任意合法数据健壮，但「53ms 的台词语义单元是否应该存在」是产品问题——若不应存在，
   需要在 `SegmentationEditOps` 上开一条最小时长约束（注意与 `4604d2b` 消灭构树期 sliver 的口径统一）

### 真机验收中发现并已修复的两个缺陷（合并在 main）

- **`afbe21c` 亚像素宽单元导致整帧绘制中断**：末单元 53ms 在 fit 缩放下宽 0.919px，
  标签 `maxWidth = 0.919 - 12 = -11.08`，`TextPainter.layout` 内部 `clampDouble` 断言
  `min <= max` 失败，异常在 `CustomPainter.paint` 中途抛出 → **本帧从抛点起的所有绘制全部丢失**：
  刻度尺 ✓ 单元轨 ✓ →抛→ 镜头轨 ✗ 抽帧 ✗ 波形 ✗ 顶栏 ✗ 底栏 ✗（Scaffold 先画 body，
  后画 appBar/bottomNavigationBar）。控件仍在树里、布局与命中测试都正常，**返回按钮点得到但看不见**。
  修法：调用点「放不下就不画标签」+ 绘制边界处 `maxWidth` 夹紧非负，双保险。
- **`9c185c8` AppLog 走 stderr 并转发框架异常**：真机以 `open`/直接跑二进制启动时，
  `debugPrint` 的输出**完全不进 stdout**，导致上面那个崩溃级异常在日志里查无痕迹（只能靠逐段插桩定位），
  与「错误不可静默吞掉」直接冲突。现改走 `stderr`，并在 `main()` 安装 `FlutterError` 转发
  （保留原处理链）。真机复验：故意指向不存在的视频，`[ishkafel][warn] 时间线抽帧失败…` 正常打出。

> 教训：**排障前先确认日志通道本身是通的**。本轮先花了大量时间在「只读模式」这条错误假说上，
> 正是因为真正的异常被日志通道吞掉了；而模式与数据在验收现场被偶然绑定（碎片单元是边界测试
> 拖出来的，测完确认切分 → 状态变 picking），进一步坐实了错误假说。最终靠「同一份含碎片数据的
> 任务在 awaitingCut / picking 下测试都红」证伪。
